// Boxbert — Etappe 4 kombiniert: NFC-Tag (I2C) startet Musik von SD.
//
// Verdrahtung:
//   SD:  CS=IO10, SCK=IO36, MISO=IO37, MOSI=IO35, VCC=USB-5V
//   NFC: SDA=IO3, SCL=IO4, VCC=3V3, DIP auf I2C (1 ON, 2 OFF)
//        Pegel gemessen 04.10.2026: SDA/SCL Ruhelage 3,3 V — direkt ok
//   I2S: BCLK=IO17, LRC=IO18, DIN=IO16 (MAX98357A-Klon, SD an 3V3, GAIN offen)
//
// Verhalten: Tag drauf  -> /happy-tune.mp3 startet (Loop bis Tag runter)
//             Tag weg   -> Stop. Neue UID wird auf Serial angezeigt.
//
// Kindersicherheit: Volume FIX auf 6 von 21 (Etappe-2-Hörtest "leise-schön").
// Kein Hochdrehen im Test. Kindergrenzwert-Cap (später ~10/21, EN 71-1)
// kommt in die Final-Firmware — hier bewusst konstant klein.
//
// Ablauf gegen Bibliotheksbeispiele geprüft:
//   Audio:  examples/I2Saudio_SD (ESP32-audioI2S v4.0.0) — braucht PSRAM=opi
//   PN532:  etappe4_nfc_i2c (04.10.2026 getestet: PN532 FW v1.6 über I2C)

#include "Arduino.h"
#include "Audio.h"
#include "SPI.h"
#include "SD.h"
#include "FS.h"
#include <Wire.h>
#include <Adafruit_PN532.h>

#define SD_CS   10
#define SPI_SCK 36
#define SPI_MISO 37
#define SPI_MOSI 35

#define I2S_BCLK 17
#define I2S_LRC  18
#define I2S_DOUT 16

#define SDA_PIN 3
#define SCL_PIN 4

constexpr uint8_t VOLUME = 6;               // 0...21 — Fix, nicht hochdrehbar
const char *PLAY_FILE = "/happy-tune.mp3";

Adafruit_PN532 nfc(255, 254, &Wire);        // IRQ/Reset offen -> Bus-Polling
Audio audio;

bool playing = false;
bool tagSeen = false;
bool autotestDone = false;

// Diagnose: die Lib meldet Gruende fuer connecttoFS-Fehler nur ueber
// diesen Callback (wie im Bibliotheksbeispiel I2Saudio_SD).
void my_audio_info(Audio::msg_t m) {
    Serial.printf("[audio] %s: %s\n", m.s, m.msg);
}

void setup() {
    Serial.begin(115200);
    delay(2500);                             // Zeitfenster fuer Serial-Monitor
    unsigned long t0 = millis();
    while (!Serial && millis() - t0 < 2000) {}

    Serial.println("=== Etappe 4 kombiniert: Tag startet Musik (I2C + SD) ===");
    Audio::audio_info_callback = my_audio_info;

    // I2C + NFC
    Wire.begin(SDA_PIN, SCL_PIN);
    Wire.setClock(100000);
    nfc.begin();
    delay(100);
    if (!nfc.getFirmwareVersion()) {
        Serial.println("FEHLER: PN532 antwortet nicht (DIP auf I2C? SDA/SCL vertauscht?)");
        while (true) delay(1000);
    }
    if (!nfc.SAMConfig()) {
        Serial.println("FEHLER: SAMConfig");
        while (true) delay(1000);
    }
    Serial.println("PN532 bereit (I2C)");

    // SD (wie Etappe 2: CS idle high, konservative Frequenz)
    pinMode(SD_CS, OUTPUT);
    digitalWrite(SD_CS, HIGH);
    SPI.begin(SPI_SCK, SPI_MISO, SPI_MOSI);
    SPI.setFrequency(1000000);
    if (!SD.begin(SD_CS)) {
        Serial.println("FEHLER: SD.begin — Karte steckt? MISO (IO37) wieder dran?");
        while (true) delay(1000);
    }
    if (!SD.exists(PLAY_FILE)) {
        Serial.printf("FEHLER: %s fehlt auf der Karte\n", PLAY_FILE);
        File root = SD.open("/");
        while (root) {
            File f = root.openNextFile();
            if (!f) break;
            Serial.printf("  vorhanden: %s\n", f.name());
            f.close();
        }
        root.close();
        while (true) delay(1000);
    }
    File probe = SD.open(PLAY_FILE);
    if (!probe) {
        Serial.printf("FEHLER: %s laesst sich nicht oeffnen\n", PLAY_FILE);
        while (true) delay(1000);
    }
    Serial.printf("SD ok — %s: %u Bytes — Heap: %u, PSRAM: %u\n",
                  PLAY_FILE, (unsigned)probe.size(),
                  (unsigned)ESP.getFreeHeap(), (unsigned)ESP.getFreePsram());
    probe.close();

    audio.setPinout(I2S_BCLK, I2S_LRC, I2S_DOUT);
    audio.setVolume(VOLUME);
    Serial.println("Warte auf Tag ...");
}

void loop() {
    audio.loop();

    // Autotest: 15 s kein Tag -> Datei einmal leise abspielen, um NFC/I2C
    // aus der Fehlersuche zu nehmen.
    if (!playing && !autotestDone && millis() > 15000) {
        autotestDone = true;
        Serial.printf("Autotest (kein Tag): spiele %s bei Volume %u\n",
                      PLAY_FILE, VOLUME);
        if (audio.connecttoFS(SD, PLAY_FILE)) {
            playing = true;
            Serial.println("Autotest: connecttoFS OK");
        } else {
            Serial.println("Autotest: connecttoFS FEHLGESCHLAGEN (s. [audio]-Zeilen)");
        }
    }

    // NFC-Poll mit kurzem Timeout, damit audio.loop() oft genug laeuft
    uint8_t uid[7] = {};
    uint8_t uidLength = 0;
    const bool tag = nfc.readPassiveTargetID(PN532_MIFARE_ISO14443A,
                                             uid, &uidLength, 50);
    if (tag) {
        if (!tagSeen) {
            Serial.print("Tag erkannt, UID: ");
            for (uint8_t i = 0; i < uidLength && i < sizeof(uid); ++i) {
                if (i) Serial.print(':');
                Serial.printf("%02X", uid[i]);
            }
            Serial.println();
            tagSeen = true;
        }
        if (!playing) {
            audio.setVolume(VOLUME);
            if (audio.connecttoFS(SD, PLAY_FILE)) {
                playing = true;
                Serial.println("Musik startet (Volume 6/21)");
            } else {
                Serial.println("FEHLER: connecttoFS");
                delay(1000);
            }
        }
    } else {
        if (tagSeen) {
            Serial.println("Tag weg — Stop");
            audio.stopSong();
            playing = false;
            tagSeen = false;
        }
    }
    delay(30);
}
