// Boxbert — Etappe 4: SD + PN532 + Audio am gemeinsamen SPI-Bus.
// Pins aus hardware/pinout.md. DIP: 1 OFF, 2 ON. NFC VCC=3V3.
// SD: CS=10, VCC=USB-5V. Beide: SCK=36, MISO=37, MOSI=35.
// NFC SS=9 (+10k Pull-up). I2S: DIN=16, BCLK=17, LRC=18.
// Erst SD in SPI-Modus bringen, dann NFC. Kein MISO/MOSI-Swap!
// Volume 6, kein lauter Testton. Nur an USB testen, Akku bleibt ab.

#include <Arduino.h>
#include <Audio.h>
#include <SPI.h>
#include <SD.h>
#include <Adafruit_NeoPixel.h>
#include <PN532_SPI.h>
#include <PN532.h>

constexpr uint8_t SD_CS = 10;
constexpr uint8_t NFC_CS = 9;
constexpr uint8_t SPI_SCK = 36;
constexpr uint8_t SPI_MISO = 37;
constexpr uint8_t SPI_MOSI = 35;
constexpr uint8_t I2S_DOUT = 16;
constexpr uint8_t I2S_BCLK = 17;
constexpr uint8_t I2S_LRC = 18;
constexpr uint8_t LED_PIN = 33; // Power=IO21 wird vom Board-Core eingeschaltet

// Die Seeed-Lib transferiert ohne eigene Transaktionen. Beim geteilten
// Bus muss JEDER Zugriff die SD-Einstellungen (MSBFIRST) durch NFC-
// Einstellungen (LSBFIRST) ersetzen und danach den Bus freigeben.
// Wrapper nur im Sketch, installierte Bibliotheken unveraendert.
class SharedPN532SPI : public PN532_SPI {
 public:
  SharedPN532SPI(SPIClass &spi, uint8_t ss) : PN532_SPI(spi, ss), bus(spi) {}
  int8_t writeCommand(const uint8_t *header, uint8_t hlen,
                     const uint8_t *body = nullptr, uint8_t blen = 0) override {
    bus.beginTransaction(SPISettings(100000, LSBFIRST, SPI_MODE0));
    const int8_t status = PN532_SPI::writeCommand(header, hlen, body, blen);
    bus.endTransaction();
    return status;
  }
  int16_t readResponse(uint8_t *buf, uint8_t len, uint16_t timeout = 1000) override {
    bus.beginTransaction(SPISettings(100000, LSBFIRST, SPI_MODE0));
    const int16_t status = PN532_SPI::readResponse(buf, len, timeout);
    bus.endTransaction();
    return status;
  }
 private:
  SPIClass &bus;
};

SharedPN532SPI transport(SPI, NFC_CS);
PN532 nfc(transport);
Audio audio;
Adafruit_NeoPixel px(1, LED_PIN, NEO_GRB + NEO_KHZ800);
const char *STARTTRACK = "/arcade-music-loop.mp3";
const char *TAGTRACK = "/subbasesoft.mp3";
const char *state = "Boot";
bool sdOk = false;
bool nfcOk = false;
bool audioReady = false;
uint8_t lastUid[7] = {};
uint8_t lastUidLength = 0;
uint32_t lastSeen = 0;

void color(uint8_t r, uint8_t g, uint8_t b) {
  px.setPixelColor(0, px.Color(r, g, b));
  px.show();
}

void my_audio_info(Audio::msg_t m) {
  Serial.printf("[audio] %s: %s\n", m.s, m.msg);
}

// Workaround fuer bestaetigten Seeed-Lib-Fehler: gueltige leere
// Konfigurationsantwort ist Laenge 0, nicht >0 (siehe Direkttest).
bool configure(const char *name, const uint8_t *command, uint8_t length) {
  const int8_t ack = transport.writeCommand(command, length);
  if (ack != 0) {
    Serial.printf("%s: ACK-Fehler %d\n", name, ack);
    return false;
  }
  uint8_t response[8] = {};
  const int16_t received = transport.readResponse(response, sizeof(response), 1000);
  Serial.printf("%s: Antwortlaenge %d (0 = OK)\n", name, received);
  return received == 0;
}

void setup() {
  digitalWrite(SD_CS, HIGH);
  pinMode(SD_CS, OUTPUT);
  digitalWrite(NFC_CS, HIGH);
  pinMode(NFC_CS, OUTPUT);
  px.begin();
  px.setBrightness(20);
  color(0, 0, 80);
  Audio::audio_info_callback = my_audio_info;

  Serial.begin(115200);
  const uint32_t started = millis();
  while (!Serial && millis() - started < 3000) delay(10);
  Serial.println("\nEtappe 4: SD ZUERST, dann NFC; gemeinsame Pins 36/37/35");
  if (!SPI.begin(SPI_SCK, SPI_MISO, SPI_MOSI)) {
    state = "FEHLER: SPI.begin";
    color(80, 0, 0);
    Serial.println(state);
    return;
  }

  sdOk = SD.begin(SD_CS, SPI, 4000000);
  Serial.println(sdOk ? "SD ok (4 MHz, SPI-Modus)" : "SD begin FEHLGESCHLAGEN");
  digitalWrite(SD_CS, HIGH);

  nfc.begin();
  delay(100);
  const uint32_t version = nfc.getFirmwareVersion();
  if (version) {
    Serial.printf("PN5%02X gefunden, Firmware %u.%u\n",
                  (unsigned)((version >> 24) & 0xFF),
                  (unsigned)((version >> 16) & 0xFF),
                  (unsigned)((version >> 8) & 0xFF));
    const uint8_t sam[] = {PN532_COMMAND_SAMCONFIGURATION, 0x01, 0x14, 0x01};
    const uint8_t retries[] = {PN532_COMMAND_RFCONFIGURATION, 0x05, 0xFF, 0x01, 0x00};
    nfcOk = configure("SAMConfig", sam, sizeof(sam)) &&
            configure("PassiveRetries", retries, sizeof(retries));
  } else {
    Serial.println("PN532 antwortet trotz SD-Init nicht — MISO-Buskonflikt weiter moeglich");
  }

  if (sdOk) {
    audioReady = audio.setPinout(I2S_BCLK, I2S_LRC, I2S_DOUT);
    if (audioReady) {
      audio.setVolume(6);
      const bool startedAudio = audio.connecttoFS(SD, STARTTRACK);
      Serial.println(startedAudio ? "Boot-Musik gestartet (Volume 6)" : "Boot-Datei nicht gestartet");
    } else {
      Serial.println("FEHLER: Audio-Pinout");
    }
  }
  if (nfcOk && sdOk && audioReady) {
    state = "SD + NFC bereit — Tag auflegen, UID + Pups testen";
    color(0, 80, 0);
  } else if (nfcOk) {
    state = "NFC bereit, SD/Audio fehlt";
    color(80, 40, 0);
  } else {
    state = "NFC fehlt — Etappe 4 NICHT bestanden";
    color(80, 40, 0);
  }
  Serial.println(state);
}

void loop() {
  if (audioReady) audio.loop();
  const uint32_t now = millis();
  static uint32_t lastReport = 0;
  if (now - lastReport >= 5000) {
    lastReport = now;
    Serial.println(state);
  }
  if (!nfcOk) {
    delay(1);
    return;
  }
  static uint32_t lastPoll = 0;
  if (now - lastPoll < 200) return;
  lastPoll = now;

  uint8_t uid[7] = {};
  uint8_t length = 0;
  // Endliche chipinterne Retries, Antworttimeout 100 ms. Zwischen den
  // Abfragen laeuft audio.loop weiter; Hoertest muss Stottern ausschliessen.
  if (!nfc.readPassiveTargetID(PN532_MIFARE_ISO14443A, uid, &length, 100)) return;
  if (length == 0 || length > sizeof(uid)) return;
  const uint32_t detected = millis();
  const bool newTag = length != lastUidLength ||
                      memcmp(uid, lastUid, length) != 0 ||
                      detected - lastSeen > 2000;
  lastSeen = detected;
  if (!newTag) return; // gehaltenes Tag darf den Pups nicht staendig neu starten
  memcpy(lastUid, uid, length);
  lastUidLength = length;
  Serial.print("Tag UID: ");
  for (uint8_t i = 0; i < length; ++i) {
    if (i) Serial.print(':');
    Serial.printf("%02X", uid[i]);
  }
  Serial.println();
  color(0, 80, 80);
  if (audioReady) {
    const bool played = audio.connecttoFS(SD, TAGTRACK);
    Serial.println(played ? "Pups gestartet — bitte auf Aussetzer achten" : "FEHLER: Pups-Datei nicht gestartet");
  }
}
