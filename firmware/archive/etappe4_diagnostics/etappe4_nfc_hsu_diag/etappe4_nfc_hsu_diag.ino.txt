// Boxbert — Etappe 4 Diagnose: PN532 HSU ohne SD, Audio oder NeoPixel.
// HSU RX=IO38 (Feather RX <- Modul SDA/TXD), TX=IO39 (Feather TX -> Modul SCL/RXD).
// Modul: DIP beide OFF, VCC=3V3, GND gemeinsam; alle SPI-Draehte vom PN532 ab.

#include <Arduino.h>
#include <PN532_HSU.h>
#include <PN532.h>

constexpr int8_t HSU_RX = 38;
constexpr int8_t HSU_TX = 39;
PN532_HSU transport(Serial1, HSU_RX, HSU_TX);
PN532 nfc(transport);

bool nfcReady = false;
uint32_t nextFirmwareProbe = 0;

bool configure(const char *name, const uint8_t *command, uint8_t length) {
  const int8_t ack = transport.writeCommand(command, length);
  if (ack != 0) {
    Serial.printf("%s: ACK-Fehler %d\n", name, ack);
    return false;
  }
  uint8_t response[8] = {};
  const int16_t received = transport.readResponse(response, sizeof(response), 1000);
  Serial.printf("%s: Antwortlaenge %d (0 = gueltige leere Antwort)\n", name, received);
  return received == 0;
}

void printVersion(uint32_t version) {
  Serial.printf("PN5%02X Firmware %u.%u\n",
                (unsigned)((version >> 24) & 0xFF),
                (unsigned)((version >> 16) & 0xFF),
                (unsigned)((version >> 8) & 0xFF));
}

bool probeFirmware() {
  const uint32_t version = nfc.getFirmwareVersion();
  if (!version) {
    Serial.println("Keine PN532-Firmware-Antwort; HSU-ACK/Antwort siehe Debugzeilen davor.");
    return false;
  }

  printVersion(version);
  const uint8_t sam[] = {PN532_COMMAND_SAMCONFIGURATION, 0x01, 0x14, 0x01};
  const uint8_t retries[] = {PN532_COMMAND_RFCONFIGURATION, 0x05, 0xFF, 0x01, 0x00};
  if (!configure("SAMConfig", sam, sizeof(sam))) return false;
  if (!configure("PassiveRetries", retries, sizeof(retries))) return false;
  Serial.println("PN532 HSU bereit — NTAG215 auflegen.");
  return true;
}

void setup() {
  Serial.begin(115200);
  const uint32_t started = millis();
  while (!Serial && millis() - started < 3000) delay(10);
  Serial.println("\nPN532 HSU Minimaldiagnose — ohne SD/Audio");
  Serial.printf("Serial1: RX=%d, TX=%d, 115200 8N1\n", HSU_RX, HSU_TX);

  nfc.begin();
  delay(300);
  nfcReady = probeFirmware();
  nextFirmwareProbe = millis() + 3000;
}

void loop() {
  if (!nfcReady) {
    if ((int32_t)(millis() - nextFirmwareProbe) >= 0) {
      nextFirmwareProbe = millis() + 3000;
      nfcReady = probeFirmware();
    }
    delay(1);
    return;
  }

  static uint32_t lastPoll = 0;
  if (millis() - lastPoll < 200) return;
  lastPoll = millis();

  uint8_t uid[7] = {};
  uint8_t uidLength = 0;
  if (!nfc.readPassiveTargetID(PN532_MIFARE_ISO14443A, uid, &uidLength, 100)) return;
  if (!uidLength || uidLength > sizeof(uid)) return;

  Serial.print("Tag UID: ");
  for (uint8_t i = 0; i < uidLength; ++i) {
    if (i) Serial.print(':');
    Serial.printf("%02X", uid[i]);
  }
  Serial.println();
}
